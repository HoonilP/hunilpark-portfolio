1. Install shiki:
   ```bash
   npm install shiki --cache /tmp/npm-cache-temp
   ```

2. Create `src/lib/shiki.ts` — singleton highlighter factory:
   ```typescript
   import { createHighlighter } from 'shiki';
   import type { Highlighter } from 'shiki';

   let highlighterPromise: Promise<Highlighter> | null = null;

   export function getHighlighter(): Promise<Highlighter> {
     if (!highlighterPromise) {
       highlighterPromise = createHighlighter({
         themes: ['github-light', 'github-dark'],
         langs: ['typescript', 'javascript', 'tsx', 'jsx', 'bash', 'python', 'json', 'css', 'sql'],
       });
     }
     return highlighterPromise;
   }
   ```

3. Create `src/components/projects/CodeBlock.tsx` — async RSC:
   ```typescript
   import { getHighlighter } from '@/lib/shiki';

   interface CodeBlockProps {
     code: string;
     lang?: string;
   }

   export default async function CodeBlock({ code, lang = 'typescript' }: CodeBlockProps) {
     const highlighter = await getHighlighter();
     const html = highlighter.codeToHtml(code, {
       lang,
       themes: {
         light: 'github-light',
         dark: 'github-dark',
       },
       defaultColor: 'light',
     });

     return (
       <div
         className="code-block my-4 overflow-x-auto rounded-lg text-sm"
         dangerouslySetInnerHTML={{ __html: html }}
       />
     );
   }
   ```

Key constraints:
- Do NOT use `codeToHtml` top-level shorthand (re-initializes highlighter per call). Use the singleton `getHighlighter()` pattern.
- Do NOT add `export const runtime = 'edge'` anywhere — shiki requires Node.js runtime.
- Use `defaultColor: 'light'` so light theme is the base, dark theme via CSS variables.
